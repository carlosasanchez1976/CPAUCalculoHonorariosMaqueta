Backend — lib/utils/puppeteerLauncher.js (esquema):
```javascript
import puppeteer from 'puppeteer-core';
import chromium from '@sparticuz/chromium';

/**
 * Lanza una instancia de Chromium optimizada para AWS Lambda
 */
export async function lanzarBrowser() {
  return puppeteer.launch({
    args: chromium.args,
    defaultViewport: chromium.defaultViewport,
    executablePath: await chromium.executablePath(),
    headless: chromium.headless,
  });
}
```

Backend — lib/services/pdfService.js (esquema):
```javascript
import { lanzarBrowser } from '../utils/puppeteerLauncher.js';
import { renderProyectoDireccionBasico } from './templates/proyectoDireccionBasico.js';

const PLANTILLAS = {
  'basico-proyecto-direccion': renderProyectoDireccionBasico,
};

export async function generarCertificado({ tipoCalculo, formData, calculationResult, calculationNumber }) {
  const renderer = PLANTILLAS[tipoCalculo];
  if (!renderer) throw new Error(`No hay plantilla para tipoCalculo: ${tipoCalculo}`);

  const html = renderer({ formData, calculationResult, calculationNumber });

  const browser = await lanzarBrowser();
  try {
    const page = await browser.newPage();
    await page.setContent(html, { waitUntil: 'networkidle0' });
    const pdfBuffer = await page.pdf({
      format: 'A4',
      printBackground: true,
      margin: { top: '10mm', right: '10mm', bottom: '10mm', left: '10mm' },
      preferCSSPageSize: true,
    });
    return pdfBuffer;
  } finally {
    await browser.close();
  }
}
```

Backend — api/v1/honorarios/exportar-pdf.js (esquema handler):
```javascript
import { generarCertificado } from '../../../lib/services/pdfService.js';

export default async function handler(req, res) {
  if (req.method !== 'POST') {
    return res.status(405).json({ success: false, error: 'Método no permitido' });
  }
  try {
    const { tipoCalculo, formData, calculationResult, calculationNumber } = req.body;
    if (!tipoCalculo || !formData || !calculationResult || !calculationNumber) {
      return res.status(400).json({ success: false, error: 'Datos incompletos' });
    }

    const inicio = Date.now();
    const pdfBuffer = await generarCertificado({ tipoCalculo, formData, calculationResult, calculationNumber });
    const duracion = Date.now() - inicio;

    console.log(`[exportar-pdf] tipoCalculo=${tipoCalculo} calc=${calculationNumber} duracion=${duracion}ms tamano=${pdfBuffer.length}B`);

    res.setHeader('Content-Type', 'application/pdf');
    res.setHeader('Content-Disposition', `attachment; filename="Honorarios-CPAU-${calculationNumber}.pdf"`);
    return res.status(200).send(pdfBuffer);
  } catch (error) {
    console.error('[exportar-pdf] error:', error);
    return res.status(500).json({ success: false, error: 'Error generando PDF' });
  }
}
```

Frontend — src/services/pdfService.js (esquema):
```javascript
const API_BASE_URL = import.meta.env.VITE_API_URL || '/api';

/**
 * Solicita al backend la generación del PDF y dispara la descarga
 */
export async function exportarPDF(payload) {
  const response = await fetch(`${API_BASE_URL}/v1/honorarios/exportar-pdf`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload),
  });

  if (!response.ok) {
    const err = await response.json().catch(() => ({}));
    throw new Error(err.error || `Error ${response.status} generando PDF`);
  }

  return response.blob();
}
```

Frontend — handleDescargarPDF refactor (esquema):
```javascript
const handleDescargarPDF = async () => {
  try {
    setDescargandoPDF(true);
    const blob = await exportarPDF({
      tipoCalculo: 'basico-proyecto-direccion',
      formData,
      calculationResult,
      calculationNumber,
    });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = `Honorarios-CPAU-${calculationNumber}.pdf`;
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    URL.revokeObjectURL(url);
  } catch (error) {
    console.error('Error descargando PDF:', error);
    alert('No pudimos generar el PDF. Por favor, intentá nuevamente.');
  } finally {
    setDescargandoPDF(false);
  }
};
```
